Промт: UK Highway Code — PWA застосунок на Django
Загальна концепція
Розробити повноцінний PWA (Progressive Web App) + вебсайт для вивчення та тестування Правил дорожнього руху Великої Британії. Контент береться виключно з офіційних джерел `gov.uk` (The Highway Code: https://www.gov.uk/guidance/the-highway-code).
Стек: Python 3.12+, Django 5.x, PostgreSQL, Redis (для Celery), Celery + celery-beat (нагадування), django-push-notifications або web-push-py (Web Push API), Tailwind CSS (через django-tailwind або CDN), Vanilla JS для PWA-логіки.

```python
class RuleSection(models.Model):
    """Розділ ПДР (наприклад: 'Rules for pedestrians', 'Using the road')"""
    title = models.CharField(max_length=255)
    slug = models.SlugField(unique=True)
    order = models.PositiveIntegerField(default=0)
    icon = models.CharField(max_length=50, blank=True)  # emoji або назва іконки
    source_url = models.URLField()  # посилання на gov.uk

    class Meta:
        ordering = ['order']

class Rule(models.Model):
    """Окреме правило / стаття ПДР"""
    section = models.ForeignKey(RuleSection, on_delete=models.CASCADE, related_name='rules')
    rule_number = models.CharField(max_length=10, blank=True)  # наприклад '1', '170'
    title = models.CharField(max_length=255, blank=True)
    content = models.TextField()  # HTML або Markdown
    has_image = models.BooleanField(default=False)
    image = models.ImageField(upload_to='rules/', blank=True, null=True)
    source_url = models.URLField()
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ['section__order', 'rule_number']
```

/static/sw.js (Service Worker)
```javascript
const CACHE_NAME = 'highway-code-v1';
const OFFLINE_URL = '/offline/';
const STATIC_ASSETS = [
  '/', '/rules/', '/quiz/', '/daily/', '/offline/',
  '/static/css/main.css', '/static/js/app.js',
  '/static/icons/icon-192.png'
];

// Install — кешуємо статику
self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE_NAME).then(cache => cache.addAll(STATIC_ASSETS))
  );
  self.skipWaiting();
});

// Activate — видаляємо старі кеші
self.addEventListener('activate', event => {
  event.waitUntil(
    caches.keys().then(keys =>
      Promise.all(keys.filter(k => k !== CACHE_NAME).map(k => caches.delete(k)))
    )
  );
  self.clients.claim();
});

// Fetch — Network First для API, Cache First для статики
self.addEventListener('fetch', event => {
  if (event.request.url.includes('/api/') || event.request.url.includes('/quiz/answer/')) {
    // Network only для динамічних запитів
    return;
  }
  event.respondWith(
    fetch(event.request)
      .catch(() => caches.match(event.request) || caches.match(OFFLINE_URL))
  );
});

// Push notifications
self.addEventListener('push', event => {
  const data = event.data?.json() || {};
  event.waitUntil(
    self.registration.showNotification(data.title || 'UK Highway Code', {
      body: data.body || "Time to practice your road rules! 🚗",
      icon: '/static/icons/icon-192.png',
      badge: '/static/icons/badge-72.png',
      data: { url: data.url || '/daily/' },
      actions: [
        { action: 'open', title: 'Study now' },
        { action: 'dismiss', title: 'Later' }
      ]
    })
  );
});

self.addEventListener('notificationclick', event => {
  event.notification.close();
  if (event.action !== 'dismiss') {
    event.waitUntil(clients.openWindow(event.notification.data.url));
  }
});
```

Web Push (нагадування)
apps/notifications/views.py
```python
from django.http import JsonResponse
from django.contrib.auth.decorators import login_required
from pywebpush import webpush, WebPushException
import json

@login_required
def subscribe(request):
    if request.method == 'POST':
        data = json.loads(request.body)
        PushSubscription.objects.update_or_create(
            endpoint=data['endpoint'],
            defaults={
                'user': request.user,
                'p256dh': data['keys']['p256dh'],
                'auth': data['keys']['auth'],
                'is_active': True,
            }
        )
        return JsonResponse({'status': 'subscribed'})

def send_push_notification(subscription, title, body, url='/daily/'):
    try:
        webpush(
            subscription_info={
                "endpoint": subscription.endpoint,
                "keys": {"p256dh": subscription.p256dh, "auth": subscription.auth}
            },
            data=json.dumps({"title": title, "body": body, "url": url}),
            vapid_private_key=settings.VAPID_PRIVATE_KEY,
            vapid_claims={"sub": f"mailto:{settings.VAPID_EMAIL}"}
        )
    except WebPushException as e:
        if '410' in str(e):  # підписка більше не дійсна
            subscription.update(is_active=False)
```
Celery task (щоденні нагадування)
```python
# apps/notifications/tasks.py
from celery import shared_task
from django.utils import timezone

@shared_task
def send_daily_reminders():
    """Запускати щоденно — розсилає нагадування у потрібний час кожного юзера"""
    now = timezone.localtime()
    subscriptions = PushSubscription.objects.filter(
        is_active=True,
        user__userprofile__push_notifications_enabled=True,
        user__userprofile__reminder_time__hour=now.hour,
        user__userprofile__reminder_time__minute__range=(now.minute - 2, now.minute + 2)
    ).select_related('user')

    daily = DailyChallenge.objects.filter(date=now.date()).first()
    for sub in subscriptions:
        send_push_notification(
            sub,
            title="🚗 Daily Highway Code Challenge!",
            body="A new question is waiting for you. Keep your streak going!",
            url='/daily/'
        )

# celery beat schedule (config/celery.py)
app.conf.beat_schedule = {
    'send-daily-reminders': {
        'task': 'apps.notifications.tasks.send_daily_reminders',
        'schedule': 60.0,  # кожну хвилину перевіряє, кому пора надсилати
    },
}
```

Анонімні користувачі
Можуть переглядати всі правила (без нотаток)
Можуть проходити тести (результати зберігаються в сесії Django)
Можуть відповідати на Daily Challenge (1 раз на день за session_key)
При спробі додати нотатку або увімкнути нагадування — soft-modal із пропозицією зареєструватись
```python
# Middleware для session_key анонімних
class AnonymousSessionMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        if not request.session.session_key:
            request.session.create()
        return self.get_response(request)
```
